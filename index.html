Полная прошивка ESP32 (Wi-Fi Access Point + HTTP Web Server + JSON API)

Готовый код для прошивки через Arduino IDE. Содержит встроенный веб-сервер и выдает JSON-пакет каждые 500 мс на 192.168.4.1. 

/*
  ==================================================================================
  PROJECT: IoT-Based Coal Mine Safety Monitoring & Real-Time Alerting System
  MICROCONTROLLER: ESP32 DevKit v1 (30-pin)
  SENSORS: MQ-135 (Gas), DHT11 (Temp/Hum), LCD 1602 I2C, Active Buzzer
  NETWORK: Built-in SoftAP (SSID: ESP32_MINE_SAFETY, IP: 192.168.4.1)
  TELEMETRY: JSON REST API endpoint "/data" updated every 500 ms
  ==================================================================================
*/

#include <WiFi.h>
#include <WebServer.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

// --- PIN DEFINITIONS (Optimized for ESP32 DevKit v1) ---
#define PIN_MQ135   34  // Analog Input (ADC1_CH6) - Never conflicts with Wi-Fi!
#define PIN_DHT     4   // OneWire Digital for DHT11
#define PIN_BUZZER  23  // Digital Output for Active Buzzer
#define PIN_SDA     21  // Hardware I2C SDA for LCD 1602
#define PIN_SCL     22  // Hardware I2C SCL for LCD 1602

#define DHTTYPE     DHT11
DHT dht(PIN_DHT, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);
WebServer server(80);

// --- WI-FI ACCESS POINT CREDENTIALS ---
const char* ssid = "ESP32_MINE_SAFETY";
const char* password = ""; // Open Access Point for instantaneous commission connection

// --- GLOBAL TELEMETRY VARIABLES ---
float gasPPM = 0.0;
float temperature = 0.0;
float humidity = 0.0;
bool isAlarm = false;
unsigned long lastSampleTime = 0;
const unsigned long SAMPLE_INTERVAL = 500; // 500 ms high-speed streaming

// Embedded Web Dashboard HTML stored in Flash memory
const char INDEX_HTML[] PROGMEM = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>ESP32 Mine Safety</title>
  <meta name="viewport" content="width=device-width,initial-scale=1">
  <style>
    body{font-family:Arial,sans-serif;background:#090d16;color:#fff;text-align:center;padding:20px}
    .card{background:#131c2e;border:1px solid #233554;border-radius:12px;padding:15px;margin:10px auto;max-width:400px}
    h2{color:#38bdf8}
    .val{font-size:32px;font-weight:bold;margin:5px 0}
    .danger{color:#ef4444}
    .safe{color:#10b981}
  </style>
</head>
<body>
  <h2>ESP32 Mine Safety Dashboard</h2>
  <div class="card"><div>Gas Level (MQ-135)</div><div id="gas" class="val">--</div><div>PPM</div></div>
  <div class="card"><div>Temperature / Humidity</div><div id="th" class="val">--</div></div>
  <div class="card"><div>Status</div><div id="st" class="val safe">NORMAL</div></div>
  \<script>
    setInterval(async () => {
      try {
        const res = await fetch('/data');
        const d = await res.json();
        document.getElementById('gas').innerText = d.gas_ppm;
        document.getElementById('th').innerText = d.temp_c + ' C / ' + d.humidity + ' %';
        
        document.getElementById('st').innerText = d.status;
        document.getElementById('st').className = 'val ' + (d.alarm ? 'danger' : 'safe');
      } catch(e) {}
    }, 500);
  <\/script>
</body>
</html>
)rawliteral";

void handleRoot() {
  server.send_P(200, "text/html", INDEX_HTML);
}

void handleData() {
  // Construct real-time JSON packet
  String json = "{";
  json += "\\"gas_ppm\\":" + String(gasPPM, 0) + ",";
  json += "\\"temp_c\\":" + String(temperature, 1) + ",";
  json += "\\"humidity\\":" + String(humidity, 0) + ",";
  
  json += "\\"buzzer\\":" + String(isAlarm ? 1 : 0) + ",";
  json += "\\"alarm\\":" + String(isAlarm ? 1 : 0) + ",";
  json += "\\"status\\":\\"" + String(isAlarm ? "CRITICAL HAZARD!" : "NORMAL") + "\\"";
  json += "}";

  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", json);
}

void setup() {
  Serial.begin(115200);

  // Initialize GPIOs
  pinMode(PIN_BUZZER, OUTPUT);
  digitalWrite(PIN_BUZZER, LOW);

  // Initialize Sensors & Peripherals
  dht.begin();
  Wire.begin(PIN_SDA, PIN_SCL);
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("MINE SAFETY SYS");
  lcd.setCursor(0, 1);
  lcd.print("Starting AP...");

  // Launch Wi-Fi Access Point (192.168.4.1)
  WiFi.softAP(ssid, password);
  IPAddress IP = WiFi.softAPIP();
  Serial.print("SoftAP IP address: ");
  Serial.println(IP);

  // Setup Web Server REST Endpoints
  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.begin();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("AP: 192.168.4.1");
  delay(1500);
  lcd.clear();
}

void loop() {
  server.handleClient();

  unsigned long currentMillis = millis();
  if (currentMillis - lastSampleTime >= SAMPLE_INTERVAL) {
    lastSampleTime = currentMillis;

    // 1. Read MQ-135 Gas Sensor
    int rawADC = analogRead(PIN_MQ135);
    gasPPM = map(rawADC, 0, 4095, 50, 1200);

    // 2. Read DHT11 Temperature & Humidity
    float t = dht.readTemperature();
    float h = dht.readHumidity();
    if (!isnan(t)) temperature = t;
    if (!isnan(h)) humidity = h;

    // 4. Determine Critical Alarm Conditions
    bool gasAlert = (gasPPM > 1000.0);
    bool tempAlert = (temperature > 28.0);
    isAlarm = (gasAlert || tempAlert);

    // 5. Trigger Physical Active Buzzer
    if (isAlarm) {
      digitalWrite(PIN_BUZZER, HIGH);
    } else {
      digitalWrite(PIN_BUZZER, LOW);
    }

    // 6. Update LCD 1602 Display
    lcd.setCursor(0, 0);
    lcd.print("G:" + String((int)gasPPM) + "ppm ");
    lcd.print(isAlarm ? "ALERT!" : "NORM  ");

    lcd.setCursor(0, 1);
    lcd.print("T:" + String((int)temperature) + "C H:" + String((int)humidity) + "% ");
    lcd.print(isAlarm ? "ALARM" : "OK   ");
  }
}
